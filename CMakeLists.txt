cmake_minimum_required(VERSION 3.21)
project(PredictiveSafeClick VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(SAFECLICK_BUILD_TESTS "Compile standalone simulation tests" OFF)
if(SAFECLICK_BUILD_TESTS)
    enable_testing()
    add_executable(safeclick_predictor_tests tests/predictor_tests.cpp)
    add_executable(safeclick_controller_tests tests/controller_tests.cpp)
    add_executable(safeclick_live_calibration_tests tests/live_calibration_tests.cpp)
    add_test(NAME predictor COMMAND safeclick_predictor_tests)
    add_test(NAME controller COMMAND safeclick_controller_tests)
    add_test(NAME live_calibration COMMAND safeclick_live_calibration_tests)
endif()

if(WIN32 AND NOT SAFECLICK_BUILD_TESTS)
    add_library(${PROJECT_NAME} SHARED src/LiveAdapter.cpp)
    if(NOT DEFINED ENV{GEODE_SDK})
        message(FATAL_ERROR "Install Geode SDK with: geode sdk install (then: geode sdk install-binaries)")
    endif()
    add_subdirectory($ENV{GEODE_SDK} "${CMAKE_CURRENT_BINARY_DIR}/geode")
    setup_geode_mod(${PROJECT_NAME})
elseif(NOT SAFECLICK_BUILD_TESTS)
    message(FATAL_ERROR "This is a Windows Geode mod. For Linux-only tests set SAFECLICK_BUILD_TESTS=ON")
endif()
